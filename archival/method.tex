\subsection{The problem with singular values}
\label{sec:singular_value_problem}

We consider two additional properties that are intuitively desirable for selecting orthonormal submatrices.
\begin{property}[Column log symmetry]
\label{property:log_symmetry}
A matrix loss $\ell$ is column log-symmetric if replacing any nonzero column $x_{.p}$ by $x_{.p}/\|x_{.p}\|_2^2$, with all other columns fixed, leaves $\ell$ unchanged.
\end{property}

\begin{property}[Column-rescaling monotonicity]
\label{property:pathwise}
A matrix loss $\ell$ is column-rescaling monotone if, for every column $x_{.d}$, $\frac{\partial\ell}{\partial\|x_{.d}\|_2}$ is negative, zero, or positive when $\|x_{.d}\|_2<1$, $\|x_{.d}\|_2=1$, or $\|x_{.d}\|_2>1$, respectively.
\end{property}

\begin{proposition}
\label{prop:inverse_symmetry_impossibility}
\label{prop:log_svd_bad}
For $D\geq2$, no loss on invertible matrices in $\mathbb R^{D\times D}$ satisfies both $\ell(X)=\ell(X^{-1})$ for every invertible $X$ and Property~\ref{property:pathwise}.
In particular, a spectral loss with $g(t)=g(t^{-1})$ for every $t>0$ cannot satisfy column-rescaling monotonicity (Property~\ref{property:pathwise}).
If $g$ is also continuously differentiable and $g'(t)>0$ for $t>1$, then $\ell$ also cannot satisfy Property~\ref{property:log_symmetry}.
\end{proposition}
Figure~\ref{fig:loss_comparison} illustrates Proposition~\ref{prop:inverse_symmetry_impossibility}.

\subsection{Losses of normalized matrices}
\label{sec:normalization}

We introduce losses satisfying Properties~\ref{property:rotation_invariance}--\ref{property:pathwise} by normalizing the columns.

\begin{definition}
\label{def:symmetric_normalization}
For a function $q:(0,\infty)\to(0,\infty)$ satisfying $q(1)=1$ and $q(t)=q(t^{-1})$, define $n_q:\mathbb R^D\setminus\{0\}\to\mathbb R^D$ by $n_q(v)=\frac{q(\|v\|_2)}{\|v\|_2}v$.
\end{definition}

We use $n_q^{\times D}(X)$ to denote column broadcasting of the normalization function, applying $n_q$ to each of the $D$ columns of $X$.
The obvious choice is $q(t)=\exp(1-\max(t,t^{-1}))$, but it is not differentiable at $t=1$.
We replace the maximum by its LogSumExp relaxation \citep{Boyd2004-ql} and rescale the resulting function so that $q(1)=1$.

\begin{proposition}
\label{prop:normalization_logsumexp}
For $t>0$, let $a(t)=\operatorname{LogSumExp}(t,t^{-1})=\log(e^t+e^{t^{-1}})$.
Normalizing $\exp(-a(t))$ to equal one at $t=1$ gives
\begin{align}
q(t)=\frac{\exp(-a(t))}{\exp(-a(1))}=\frac{2e}{e^t+e^{t^{-1}}}.
\label{eq:normalization}
\end{align}
\end{proposition}

\begin{proposition}
For $q$ defined in \eqref{eq:normalization}, $-\log q$ is convex, $q$ is smooth on $(0,\infty)$, and $\log q(t)$ decays linearly in $t$ as $t\to\infty$ and in $t^{-1}$ as $t\to0$.
\end{proposition}

Many matrix functions applied to suitably normalized matrices are uniquely minimized by orthonormal matrices.
Two examples are the inverse determinant and the Frobenius norm of the inverse.
Figure~\ref{fig:loss_comparison} illustrates the loss characterizations in Table~\ref{tab:loss_functions}.
\begin{proposition}[Normalized spectral losses]
\label{prop:normalized_spectral_losses}
For full-rank $X\in\mathbb R^{D\times D}$, let $g:(0,\infty)\to\mathbb R$ be twice differentiable with $g(1)=0$, $g'(t)<0$, and $t g''(t)>g'(t)$ for every $t>0$.
Then $\ell_g(X)=\sum_{d=1}^D g\!\left(\sigma_d(n_q^{\times D}(X))\right)$ is nonnegative and satisfies Properties~\ref{property:rotation_invariance}--\ref{property:pathwise}.
\end{proposition}
\begin{proposition}[Inverse determinant]
\label{prop:determinant_singular_values}
For $g(t)=-\log t$, the centered inverse determinant satisfies $|\det(n_q^{\times D}(X))|^{-1}-1=\exp(\ell_g(X))-1$.
\end{proposition}
\begin{proposition}[Inverse Frobenius norm]
\label{prop:frobenius_inverse_singular_values}
For $g(t)=t^{-2}-1$, the centered inverse Frobenius norm satisfies $\|(n_q^{\times D}(X))^{-1}\|_F-\sqrt D=\sqrt{\ell_g(X)+D}-\sqrt D$.
\end{proposition}

We have established that there are losses $\ell$ that satisfy our conditions for a good loss function on square matrices $X \in \mathbb R^{D\times D}$.
However, the computational identification of such matrices is combinatorial in nature.
To avoid this, we reformulate the problem as a sparse convex optimization problem.


\subsection{Isometry pursuit}

Isometry pursuit is the application of multitask basis pursuit to the normalized design matrix $w(X)$ to identify submatrices of $ X$ that are as orthonormal as possible.
For $\beta\in\mathbb R^{P\times D}$, define $\|\beta\|_{1,2}=\sum_{p=1}^P\|\beta_{p.}\|_2$.
The set of isometry pursuit solutions is
\begin{align}
\label{prog:isometry_pursuit}
\mathcal B(X):=\arg\min_{\beta\in\mathbb R^{P\times D}}\|\beta\|_{1,2}\;:\;I_D=w(X)\beta.
\end{align}

Recovered functions are the indices of the dictionary elements with non-zero coefficients.
That is, they are given by $S(\beta)$, where $S:\mathbb R^{P\times D}\to 2^{[P]}$ is defined by $S(\beta)=\{p\in[P]:\|\beta_{p.}\|_2>0\}$.
\begin{algorithm}[H]
\caption{\isometrypursuit(Matrix ${X} \in \mathbb{R}^{D \times P}$)}
\begin{algorithmic}[1]
\STATE {Normalize} $X' = w({X})$
\STATE {Optimize} $\widehat \beta \in \mathcal B(X)$
\STATE {\bf Output} $\widehat{S} = S (\widehat \beta)$
\end{algorithmic}
\end{algorithm}

The multitask basis pursuit solution need not be unique \citep{JMLR:v23:21-0420} or cardinality $D$.
However, Proposition~\ref{prop:unitary_selection} guarantees that the union of supports of all minimizers contains any orthonormal subset.
Since the recovered support need not have cardinality $D$, we first apply \isometrypursuit~ to prune the $P$ candidate features, then apply \textsc{Greedy} to the reduced dictionary.
The two-stage approach is standard in the Lasso literature \citep{Meinshausen2007-ey, Hesterberg2008-iy, Koelle2022-ju, Bruckmeier2023-cube}.
Here \textsc{Greedy} denotes \greedy~or \backwardgreedy~followed by \forwardbackward, as described in Section~\ref{sec:sparse_convex_coordinate_selection}.
In practice, selection of $\beta\in\mathcal B(X)$ depends on the optimizer (we use SCS \citep{O-Donoghue2013-wf}).
We empirically assess the supports and greedy searches obtained numerically in Section~\ref{sec:experiments}.

\begin{algorithm}[H]
\caption{\tsip(Matrix ${X} \in \mathbb{R}^{D \times P}$)}
\begin{algorithmic}[1]
\STATE $\widehat{S}_{IP} = \text{\isometrypursuit}(X)$
\STATE $\widehat{S} = \text{\textsc{Greedy}}({X}_{.\widehat{S}_{IP}}, \ell, D)$ \COMMENT{Return original column indices.}
\STATE {\bf Output} $\widehat{S}$
\end{algorithmic}
\end{algorithm}

\subsection{Theory}

The intuition behind our application of multitask basis pursuit is that submatrices consisting of vectors which are closer to 1 in length and more orthogonal will have smaller loss.
The collection of optimal supports is invariant to the orthonormal basis used for $X$.
\begin{proposition}
\label{prop:basis_pursuit_selection_invariance}
Let $U \in \mathbb R^{D \times D}$ be orthonormal.
Then $\mathcal B(UX)=\{\beta U^\top:\beta\in\mathcal B(X)\}$.
\end{proposition}
Thus the collection of optimal supports is independent of the orthonormal basis.

\begin{proposition}
\label{prop:unitary_selection}
Let $X\in\mathbb R^{D\times P}$ contain an orthonormal column-submatrix $X_{.S}$.
Then $S\subseteq\bigcup_{\beta\in\mathcal B(X)}S(\beta)$.
\end{proposition}

The orthonormal subset supports a minimizer, but a numerical minimizer need not contain the union of supports.

\subsubsection{Non-uniqueness}

The isometric solution is optimum among cardinality $D$ solutions.
\begin{proposition}
\label{prop:D_min}
$\ell (X_S) = D , |S| = D \Leftrightarrow X_S X_S^T = I_D$
\end{proposition}
\begin{proposition}
\label{prop:loss_without_isometry}
Without $|S|=D$, $\ell(X_{.S})=D$ does not imply that $X_{.S}$ contains an orthonormal column-submatrix.
\end{proposition}

\subsubsection{Computational complexity}

\begin{proposition}
\label{prop:pseudoinverse_cost}
For $X\in\mathbb R^{D\times P}$, evaluating $\|X^\dagger\|_{1,2}$ costs $C_\ell=O(DP\min(D,P))$, which is $O(D^2P)$ when $P\geq D$.
\end{proposition}
The above proposition follows immediately from the computational complexity of singular value decomposition \citep{Liang2016-qb}.
\begin{proposition}
\label{prop:computational_complexity}
The computational complexity of each step of Isometry Pursuit is $O(D^2P)$.
\end{proposition}

For $\mathcal S$ iterations of \isometrypursuit, the arithmetic-work proxy is $\mathcal S D^2P$.
We use the observed iteration count when comparing this proxy with runtime in Figure~\ref{fig:complexity-runtime-early-stop}.
For \greedy~and \backwardgreedy, we sum the loss-evaluation costs over all candidate additions or deletions.
We solve the convex optimization problem using SCS \citep{ocpb:16} through CVXPY \citep{diamond2016cvxpy, agrawal2018rewriting}.
