Our goal is, given a matrix $ X \in \mathbb R^{D \times P}$, to select a subset $ S \subset [P]$ with $| S| = D$ such that $X_{.  S}$ is as orthonormal as possible in a computationally efficient way.
Figures \ref{fig:data} and \ref{fig:selection} provide a visual depiction of this problem.
We motivate this problem from the perspective of non-linear embeddings.

\subsection{Isometry}

Isometries preserve distances and angles, and so can be used faithfully for subsequent human or algorithmic analysis \citep{ZhangZ:04,ChenBuja:localMDS09,tenenbaum2000ggf}.

\begin{definition}{\citet{golub2013matrix}}
\label{def:isometry}
A matrix $X \in \mathbb R^{D \times D}$ is an \textbf{isometry} or, equivalently, \textbf{unitary} if $X X^T = I_D$.

\end{definition}

\begin{proposition}{\citet{golub2013matrix}}
\label{prop:orthonormal_spectrum}
The singular values $\sigma_1(X) \dots \sigma_D(X)$ are equal to $1$ if and only if $X$ is an isometry.
\end{proposition}
\begin{proposition}{\citet{golub2013matrix}}
\label{prop:orthonormal_basis}
If $P = D$, the component vectors $X_{.1} \dots X_{.D} \in \mathbb R^D$ form an isometry if and only if, for all $d, d' \in [D], \langle X_{.d}, X_{.d'} \rangle = \begin{cases} 1 \; d = d' \\ 0 \; d \neq d'
\end{cases}$.
\end{proposition}

\begin{definition}{\citet{Lee2018-rm}}
The \textbf{differential} of a smooth map $\phi:\mathcal M \to \mathcal N$ between $D$ dimensional manifolds $\M \subseteq \mathbb R^B$ and $\N \subseteq \mathbb R^P$ is a map in orthonormal tangent bases $x_1 \dots x_{D}$ of $T_\xi \M$ and $y_1 \dots y_{D}$ of $T_{\phi(\xi)} \N$ consisting of entries $D\phi(\xi)_{ij} = \langle y_i, \frac{\partial \phi}{\partial x_j} \rangle (\xi)$.
% Source locator: Lee2018-rm, Appendix A, p. 377.
\end{definition}

\begin{definition}{\citet{Lee2018-rm}}
\label{def:isometric_at_a_point}
A map $\phi$ between $D$ dimensional submanifolds is an \textbf{isometry at a point} $\xi \in \mathcal M$ if $D \phi (\xi)$ is an isometry.
\end{definition}

\begin{definition}{\citet{Lee2018-rm}}
\label{def:conformal_at_a_point}
A map $\phi$ between $D$ dimensional submanifolds is \textbf{conformal at a point} $\xi \in \mathcal M$ if there exists a scalar $\lambda(\xi) > 0$ such that ${D \phi (\xi)}^T D \phi (\xi) = \lambda(\xi) I_D$.
\end{definition}

\begin{proposition}{\citet{Lee2018-rm}}
\label{prop:distance_differential_isometry}
Let $\phi:\mathcal M \to \mathcal N$ be a bijection between connected $D$-dimensional Riemannian submanifolds.
Then $d_\mathcal N(\phi(\xi), \phi(\xi')) = d_\mathcal M(\xi, \xi')$ if and only if $\phi$ is smooth and is an isometry at every point.
% Source locator: Lee2018-rm, Problem 6-7.
\end{proposition}

\subsection{Coordinate selection}
\label{sec:sparse_convex_coordinate_selection}

Best subset selection algorithms for selecting $D$ coordinates from a dictionary $f_{1:P}$ of candidate features vary along several axes\citep{Kohli2021-lr, Koelle2022-ju, Koelle2024-no}.
Depending on the loss function used for selection, we can select coordinates that target various metric properties like isometry, conformality, or other well-conditionedness.
Loss functions can input manifold distances or tangent space differentials, although within an asymptotically shrinking neighborhood, this distinction becomes less important.
Gradients can be computed analytically or numerically, and the loss can be evaluated at a single point or over an entire dataset.
Regardless, the dictionary-based coordinate search problem best subset selection problem is
\begin{align}
\label{prog:function_best_subset_selection}
\arg \min_{ S \in \binom{[P]}{D}} \ell ( f_{. S})
\end{align}
where $\ell$ is some loss function that measures isometricness of a selected functional subset $f_{. S}$.

Greedy algorithms are a standard approach to problems of the form~\eqref{prog:function_best_subset_selection}, but need not find a global minimum.
\brute~finds a global minimum at cost $O(C_\ell\binom{P}{D})$, where $C_\ell$ is the cost of evaluating $\ell$.
\greedy~adds the column giving the smallest resulting loss until $D$ columns remain selected; \backwardgreedy~starts with all columns and removes the column giving the smallest resulting loss until $D$ remain.
\forwardbackward~then takes the best improving selected--unselected column exchange until no exchange reduces the loss.
Formal algorithms are given in Section~\ref{sec:algorithms}.

Convex relaxation provides another approach to subset selection \citep{Liberti2004-iu, Abdi2013-th, Zhou2021-zz}.
For example, consider the overcomplete regression problem \citep{Chen2001-hh}
\begin{align}
\arg \min_{\beta \in \mathbb R^P : y  = X\beta} \ell(\beta),
\end{align}
where $P>D$ and the solution $\beta$ is not unique.
When $\ell(\beta) = \|\beta\|_0$, this non-convex problem searches directly for a sparsest representing subset.
However, when $\ell(\beta) =\|\beta\|_1$, the problem is a convex relaxation and may be solved efficiently.
Because the $\ell_1$ objective is scale-sensitive, this relaxation is geometrically meaningful for coordinate selection only after normalizing the columns of $X$ so that sparsity is not confounded with column length.

\subsection{Isometry losses}

One family of losses consists of linearized objectives that apply matrix loss functions to the differential $Df_{.S}(\xi)$.
For $D$ selected functions, isometry at $\xi$ is equivalent to orthonormality of the matrix $Df_{.S}(\xi)$.

\begin{property}[Rotation invariance]
\label{property:rotation_invariance}
A matrix loss $\ell$ is rotation invariant if, for every orthonormal $U \in \mathbb R^{D \times D}$, $\ell(UX) = \ell(X)$.
\end{property}

\begin{property}[Unique isometry minimum]
\label{property:unique_isometry_minimum}
A matrix loss $\ell$ is uniquely minimized by isometries if $\arg \min_X \ell(X) = \{X : XX^T = I_D\}$.
\end{property}

\begin{definition}{\citet{Sra2011-sdivergence}}
The \textbf{S-divergence} $D(P,P')$ on positive definite matrices $P,P'$ is given by $D(P,P') = (\log \det (\frac{P + P'}{2}) - \frac{1}{2}\log \det(PP'))^{\frac{1}{2}}$.
\end{definition}
\begin{proposition}
\label{prop:sdiv_singular_values}
If $X^T X$ is positive definite, then $D(X^T X,I_D)^2 = \sum_{d=1}^D \log \frac{\sigma_d(X) + \sigma_d(X)^{-1}}{2}$.
\end{proposition}
\begin{proposition}
\label{prop:sdiv_minimized_by_isometries}
If $X^T X$ is positive definite, then $D(X^T X,I_D)^2$ is uniquely minimized by isometric matrices.
\end{proposition}
See \citep{Cichocki2014-logdet} for a more extensive discussion of other matrix divergences.
