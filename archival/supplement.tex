\subsection{Algorithms}
\label{sec:algorithms}

We give definitions of the brute and greedy algorithms for the combinatorial problem studied in this paper.
The brute force algorithm is computationally intractable for all but the smallest problems, but always finds the global minima.

\begin{algorithm}[H]
\caption{\brute(Matrix ${X} \in \mathbb{R}^{D \times P}$, objective $f$)}
\begin{algorithmic}[1]
\FOR{each combination $S \subseteq \{1, 2, \dots, P\}$ with $|S| = D$}
    \STATE Evaluate $f({X}_{.S})$
\ENDFOR
\STATE {\bf Output} the combination $S^*$ that minimizes $f({X}_{.S})$
\end{algorithmic}
\end{algorithm}

Greedy algorithms are computationally expedient but can get stuck in local optima \citep{Cormen, Russell-09}, even with randomized restarts \citep{Dick2014HowMR}.

\begin{algorithm}[H]
\caption{\greedy(Matrix ${X} \in \mathbb{R}^{D \times P}$, objective $f$, selected set $S = \emptyset$, current size $d=0$)}
\begin{algorithmic}[1]
\IF{$d = D$}
    \STATE {\bf Return} $S$
\ELSE
    \STATE {\bf Initialize} $S_{\text{best}} = S$
    \STATE {\bf Initialize} $f_{\text{best}} = \infty$
    \FOR{each $p \in \{1, 2, \dots, P\} \setminus S$}
        \STATE {\bf Evaluate} $f({X}_{.(S \cup \{p\})})$
        \IF{$f({X}_{.(S \cup \{p\})}) < f_{\text{best}}$}
            \STATE {\bf Update} $S_{\text{best}} = S \cup \{p\}$
            \STATE {\bf Update} $f_{\text{best}} = f(\mathcal{X}_{.(S \cup \{p\})})$
        \ENDIF
    \ENDFOR
    \STATE {\bf Return} \greedy(${X}$, $f$, $S_{\text{best}}$, $d+1$)
\ENDIF
\end{algorithmic}
\end{algorithm}

\subsubsection{Proof of Proposition \ref{prop:sdiv_singular_values}}
\label{proof:sdiv_singular_values}

\begin{proof}
First,
\begin{align}
\frac{1}{2}\log \det(X^T X) &= \frac{1}{2} \log \prod_{d=1}^D \sigma_d(X)^2 \\
&= \log \prod_{d=1}^D \sigma_d(X) \\
&= \sum_{d=1}^D \log \sigma_d(X).
\end{align}
Secondly, $I_D$ and $X^T X$ are diagonalizable in the same basis, so
\begin{align}
\log \det (\frac{X^T X + I_D}{2}) &= \log \prod_{d=1}^D \frac{\sigma_d(X)^2 + 1}{2} \\
&= \sum_{d=1}^D \log \frac{\sigma_d(X)^2 + 1}{2}.
\end{align}
Then,
\begin{align}
D(X^T X,I_D)^2
&= \sum_{d=1}^D (\log \frac{\sigma_d(X)^2 + 1}{2} - \log \sigma_d(X)) \\
&= \sum_{d=1}^D \log \frac{\sigma_d(X)^2 + 1}{2 \sigma_d(X)} \\
&= \sum_{d=1}^D \log \frac{\sigma_d(X) + \sigma_d(X)^{-1}}{2}.
\end{align}
\end{proof}

\subsubsection{Singular values of nearly colinear vectors}
\label{sec:nearly_colinear_singular_values}

We now exhibit a matrix for which no $g$ can be chosen such that a log det divergence satisfies property. 
We parametrize this matrix as
\begin{align}
X_\epsilon(r) =
\begin{bmatrix}
1 & r (1 - \epsilon^2)^{1/2} \\
0 & r \epsilon
\end{bmatrix}
\end{align}
where $r$ is the length of the second vector and $0<\epsilon<1$ controls the angle between the two vectors.
We compute
\begin{align}
\frac{\partial \ell}{\partial r} |_{r=1} &= \frac{\partial }{\partial r} (g(\sigma_1(X_\epsilon(r))) + g(\sigma_2(X_\epsilon(r)))) |_{r=1} \\
&= g'(\sigma_1(X_\epsilon(r)))\frac{\partial }{\partial r} \sigma_1(X_\epsilon(r)) + g'(\sigma_2(X_\epsilon(r)))\frac{\partial }{\partial r} \sigma_2(X_\epsilon(r)) 
\end{align}
We will to show that, for any log-symmetric $g$ and any $\epsilon != 0$
\begin{align}
g'(\sigma_1(X_\epsilon(r)))\frac{\partial }{\partial r} \sigma_1(X_\epsilon(r)) |_{r = 1} \neq  - g'(\sigma_2(X_\epsilon(r)))\frac{\partial }{\partial r} \sigma_2(X_\epsilon(r)) |_{r = 1}  
\end{align}

First, let's compute $\sigma_1(X_\epsilon(r)) $ and $\sigma_2(X_\epsilon(r))$.
\begin{align}
X_\epsilon(r)^T X_\epsilon(r)
=
\begin{bmatrix}
1 & r (1 - \epsilon^2)^{1/2} \\
r (1 - \epsilon^2)^{1/2} & r^2
\end{bmatrix}.
\end{align}
Recall that $\lambda$ is an eigenvalue of a matrix $A$ if there is a non-zero vector $v$ such that $Av=\lambda v$.
Equivalently,
\begin{align}
\det(A - \lambda I) = 0.
\end{align}
In our case, this gives
\begin{align}
0
&=
\det
\begin{bmatrix}
1 - \lambda & r (1 - \epsilon^2)^{1/2} \\
r (1 - \epsilon^2)^{1/2} & r^2 - \lambda
\end{bmatrix} \\
&= (1-\lambda)(r^2-\lambda) - r^2(1-\epsilon^2) \\
&= \lambda^2 - (1+r^2)\lambda + \epsilon^2 r^2.
\end{align}
Thus the eigenvalues of $X_\epsilon(r)^T X_\epsilon(r)$ are
\begin{align}
\lambda_1(r)
&=
\frac{1+r^2 + \sqrt{(1+r^2)^2 - 4\epsilon^2 r^2}}{2} \\
\lambda_2(r)
&=
\frac{1+r^2 - \sqrt{(1+r^2)^2 - 4\epsilon^2 r^2}}{2}.
\end{align}
Since singular values are the square roots of the eigenvalues of $X_\epsilon(r)^T X_\epsilon(r)$,
\begin{align}
\sigma_1(X_\epsilon(r))
&=
\left[
\frac{1+r^2 + \sqrt{(1+r^2)^2 - 4\epsilon^2 r^2}}{2}
\right]^{1/2} \\
\sigma_2(X_\epsilon(r))
&=
\left[
\frac{1+r^2 - \sqrt{(1+r^2)^2 - 4\epsilon^2 r^2}}{2}
\right]^{1/2}.
\end{align}

Differentiating the singular values gives
\begin{align}
\frac{\partial \sigma_1}{\partial r} &= \frac{1}{2}(\frac{1 + r^2 + \sqrt{(1 + r^2)^2-4\epsilon^2r^2}}{2})^{-1/2} \frac{2r + \frac{1}{2} (((1+r^2)^2 - 4\epsilon^2r^2)^{-1/2}(2(1+r^2)(2r) - 8\epsilon^2r))}{2} \\
\frac{\partial \sigma_2}{\partial r}
&=
\frac{1}{2}
\left(
\frac{1 + r^2 - \sqrt{(1 + r^2)^2-4\epsilon^2r^2}}{2}
\right)^{-1/2}
\frac{
2r
-
\frac{1}{2}
\left(((1+r^2)^2 - 4\epsilon^2r^2)^{-1/2}
\left(2(1+r^2)(2r) - 8\epsilon^2r\right)\right)
}{2}.
\end{align}
Then
\begin{align}
\sigma_1(X_\epsilon(1)) &= (1 + \sqrt{1 -\epsilon^2} )^{1/2}\\
\sigma_2(X_\epsilon(1)) &= (1 - \sqrt{1 -\epsilon^2} )^{1/2} \\
\frac{\partial \sigma_1}{\partial r} |_{r=1}&= \frac{1}{2}
\left(1 + \sqrt{1 - \epsilon^2}\right)^{1/2}\\
\frac{\partial \sigma_2}{\partial r} |_{r=1}&= \frac{1}{2}
\left(1 - \sqrt{1 - \epsilon^2}\right)^{1/2}
\end{align}
Now we can specify $g$.
Suppose
\begin{align}
g'\left(\left(1+\sqrt{1-\epsilon^2}\right)^{1/2}\right)
\frac{1}{2}
\left(1+\sqrt{1-\epsilon^2}\right)^{1/2}
=
-
g'\left(\left(1-\sqrt{1-\epsilon^2}\right)^{1/2}\right)
\frac{1}{2}
\left(1-\sqrt{1-\epsilon^2}\right)^{1/2}.
\end{align}






The second derivative is easier to obtain from the determinant.
Since
\begin{align}
\det(X_\epsilon(r)^T X_\epsilon(r))
&= r^2 - r^2(1-\epsilon^2) \\
&= \epsilon^2r^2,
\end{align}
and since this determinant is also $\sigma_1(r)^2\sigma_2(r)^2$,
\begin{align}
\sigma_1(r)\sigma_2(r)=\epsilon r
\end{align}
since singular values are nonnegative.
Differentiating both sides gives
\begin{align}
\sigma_2(r)\frac{\partial\sigma_1}{\partial r}
+
\sigma_1(r)\frac{\partial\sigma_2}{\partial r}
&= \epsilon.
\end{align}
Therefore,
\begin{align}
\frac{\partial \sigma_2}{\partial r}
&=
\frac{\epsilon}{\sigma_1(r)}
-
\frac{\sigma_2(r)}{\sigma_1(r)}
\frac{\partial \sigma_1}{\partial r} \\
&=
\frac{\epsilon}{\sigma_1(r)}
-
\frac{\epsilon r}{\sigma_1(r)^2}
\frac{\partial \sigma_1}{\partial r}.
\end{align}
For a singular value loss $\ell(X)=\sum_d g(\sigma_d(X))$,
\begin{align}
\frac{\partial \ell(X_\epsilon(r))}{\partial r}
&=
g'(\sigma_1(r))\frac{\partial \sigma_1}{\partial r}
+
g'(\sigma_2(r))\frac{\partial \sigma_2}{\partial r}.
\end{align}
At $r=1$, these expressions simplify to
\begin{align}
\sigma_1(1)^2 &= 1 + (1-\epsilon^2)^{1/2} \\
\sigma_2(1)^2 &= 1 - (1-\epsilon^2)^{1/2}
\end{align}
and
\begin{align}
\frac{\partial \sigma_1}{\partial r}\bigg|_{r=1}
&= \frac{\sigma_1(1)}{2} \\
\frac{\partial \sigma_2}{\partial r}\bigg|_{r=1}
&= \frac{\sigma_2(1)}{2}
= \frac{\epsilon}{2\sigma_1(1)}.
\end{align}
Therefore,
\begin{align}
\frac{\partial \sigma_1 / \partial r}{\partial \sigma_2 / \partial r}\bigg|_{r=1}
&=
\frac{\sigma_1(1)}{\sigma_2(1)} \\
&=
\frac{1 + (1-\epsilon^2)^{1/2}}{\epsilon}.
\end{align}
This ratio diverges as $\epsilon \to 0$.
Thus, as the vectors become more colinear, changing the length of the second vector has an arbitrarily larger first-order effect on the large singular value than on the small singular value.

\subsubsection{Proof of Proposition \ref{prop:basis_pursuit_selection_invariance}}
\label{proof:basis_pursuit_program_invariance}

In this proof we first show that the penalty $\|\beta\|_{1,2}$ is unchanged by unitary transformation of $\beta$.

 \begin{proposition}
 \label{prop:basis_pursuit_loss_equivalence}
 Let $U \in \mathbb R^{D \times D}$ be unitary.
 Then $\|\beta\|_{1,2} = \|\beta U \|$.
\end{proposition}

\begin{proof}
\begin{align}
\|\beta U \|_{1,2} &= \sum_{p = 1}^P \| \beta_{p.} U \| \\
&= \sum_{p = 1}^P \| \beta_{p.} \| \\
&= \|\beta \|_{1,2}
\end{align}
\end{proof}

We then show that this implies that the resultant loss is unchanged by unitary transformation of $ X$.

\begin{proposition}
 \label{prop:basis_pursuit_loss_equivalence}
 Let $U \in \mathbb R^{D \times D}$ be unitary.
 Then $\widehat \beta  (U  X) = \widehat \beta  (  X) U$.
\end{proposition}

\begin{proof}
\begin{align}
\widehat \beta  (U  X)  &= \arg \min_{\beta \in \mathbb R^{P \times D}} \|\beta\|_{1,2}  \; : \; I_{D} = U X \beta \\
&= \arg \min_{\beta \in \mathbb R^{P \times D}} \|\beta\|_{1,2}  \; : \; U^{-1} U = U^{-1} U X \beta U \\
&= \arg \min_{\beta \in \mathbb R^{P \times D}} \|\beta\|_{1,2}  \; : \;  I_D = X \beta U \\
&= \arg \min_{\beta \in \mathbb R^{P \times D}} \|\beta U \|_{1,2}  \; : \;  I_D = X \beta U \\
&= \arg \min_{\beta \in \mathbb R^{P \times D}} \|\beta \|_{1,2}  \; : \;  I_D = X \beta.
\end{align}
\end{proof}

\subsubsection{Proof of Proposition \ref{prop:unitary_selection}}
\label{proof:unitary_selection}

In this proof we first show that the orthonormal submatrix $X_{.S}$ is contained within the set $\widehat { \mathcal \beta}_{MBP} (w(X), I_D)$.

\begin{proposition}
\label{prop:kkt}
Let $X_{.S}$ be a rank $D$ orthonormal submatrix of $X$.
Then $S \in S(\beta) : \beta \in \widehat{ \mathcal \beta}_{MBP} (w(X), I_D))$.
 \end{proposition}
 
 \begin{proof}
 
 By Proposition \ref{prop:basis_pursuit_selection_invariance} without loss of generality, let $X_{.S} = I_D$.
 Also, without loss of generality let $S = \{1 \dotsc D\}$.
 We show that $\beta = [I_D \; 0]$ satisfies the KKT conditions for $X = [I_D \; X_{-S}]$.
 
Recall that the KKT conditions for a convex optimization problem are a set of conditions on an element of the domain of the optimization algorithm that, if satisfied, certify that the element is a solution.
To write out the KKT conditions, we define the Lagrangian
\begin{align}
\mathcal L(X,\beta,\nu) = \|\beta\|_{1,2} + \nu^T (I_D - w(X) \beta)
\end{align}
where $\nu \in \mathbb R^{D\times D}$ is the dual variable.

The KKT conditions are then
\begin{itemize}
\item Primal feasibility: $w(X) \beta = I_D$
\item Stationarity: there exists a dual variable $\nu$ such that $0 \in \partial \|\beta\|_{1,2} - w(X)^T \nu$  where $\partial$ is the subdifferential operator.
\end{itemize}
where dual feasibility and complementary slackness are ignorable by virtue of the absence of inequality constraints.

First note that in our case, primal feasibility is satisfied by $X$ being rank $D$ and $w$ not effecting the rank of the transformed matrix since it only rescales length and not direction.

For stationarity, recall that 
\begin{align}
\|\beta\|_{1,2} = \sum_{p = 1}^P \|\beta_{p.}\|_2.
\end{align}
Then, by the definition of subdifferential
\begin{align}
\partial \|\beta\|_{1,2} = \begin{bmatrix}
I_D \\
v_{d+1} \\
\dotsc \\
v_P
\end{bmatrix}
\end{align}
where $v_p \in \mathbb R^{D}$ can be any vector satisfying $\|v_p\|_2 \leq 1$.

We therefore must show that there exists a $\nu$ that satisfies
\begin{align}
\begin{bmatrix}
I_D \\
w(X_{-S})^T
\end{bmatrix}\nu  = \begin{bmatrix}
I_D \\
V_{-S}
\end{bmatrix}.
\end{align}
At this point we can see that in fact $I_D$ is an appropriate choice of $\nu$ since normalization by $w$ leads to vectors satisfying $\|w(X)\|_2 \leq 1$.

Since for this $\beta$, $S(\beta) = S$, we have shown that $S$ is the support of a solution to $\beta_{MBP} (w(X), I_D)$.
\end{proof}

The next part of the proof handles the presence of non-unique solutions.

\begin{proposition}
\label{prop:support_containment}
$S(\widehat \beta ( X)) = \cup \; S(\beta) \; : \;  \beta \in  \mathcal \beta_{MBP} (w(X), I_D)$
 \end{proposition}

\begin{proof}
The proposition follows from geometric properties of the solution set.
We ignore the case for singleton solutions, for which the proposition is self-evident.
Recall that $\widehat \beta ( X) := \widehat \beta_{MBP}^{\ell_2} ( w(X), I_D ) =  \arg \min_{\beta \in \mathbb R^{P \times D} } \|\beta\|_F : \beta \in \widehat {\mathcal \beta}_{MBP}(X, I_D)  $.
By definition of $\widehat {\mathcal \beta}_{MBP} (X, I_D) $, $\|\beta_{p.} \|_2 : \beta \in  \widehat { \mathcal \beta}_{MBP} (X, I_D) $ is an affine linear subspace of $\mathbb R^{P}.$
The projection of the vector of minimum $\ell_2$ distance from the origin to a affine space intersects that space perpendicularly.
However, for all $\|\beta_{.p}\|_2$ such that $\beta$ is on the intersection of this affine space with the coordinate hyperplanes formed by fixing any choices of $\|\beta_{.p}\|_2 = 0$, this vector doesn't intersect this space perpendicularly.
Thus, $\widehat \beta ( X)$ is not on the intersection of the solution polytope $ \widehat {\mathcal \beta}_{MBP} (w(X), I_D)$ with the coordinate hyperplanes.
It is therefore maximally non-sparse with respect to any of the minimizing solutions.
\end{proof}

\subsubsection{Proof of Proposition \ref{prop:D_min}}

$X_S$ admits a QR decomposition $X_S = QR$ where $Q$ is orthogonal and $R$ is upper triangular.
By Proposition \ref{prop:basis_pursuit_selection_invariance}, we can assume without loss of generality that $Q = I_D$.
That is, $X_S$ is upper triangular.
Thus, $\beta = X_S^{-1}$ is upper triangular with diagonal elements $\beta_{dd} = R_{dd}^{-1}$.
$\|\beta_{d.}\|_2 \geq \beta_{dd}$, with equality only occurring when the off-diagonal elements are $0$.
Therefore, $\|\beta_{1,2}\| \geq D$ with equality occurring only when $\beta = I_D$.

\subsubsection{Proof of Proposition \ref{prop:computational_complexity}}
\label{proof:computational_complexity}

\begin{proof}
    The augmented split Lagrangian minimized by ADMM is
    \begin{align}
    \mathcal L_{\rho,\gamma, \nu} (X,\beta,z) = \|z\|_{1,2} + \text{tr}(\nu^T (I_D - w(X) \beta)) + \rho \|I_D - w(X) \beta\|_F^2 + \gamma \|\beta - z\|_F^2
    \end{align}
    This minimization problem is
    \begin{align}
        \arg \min_{\beta,z \in P \times D} \mathcal L_{\rho,\gamma,\nu} (X,\beta,z)
    \end{align}
    For a more detailed explanation of the reasons to prefer optimizing the split augmented Lagrangian over direct optimization of Equation \ref{eq:bp} see \citet{Boyd2010-jd}.
    
    The three main subroutines ( \ref{eq:beta_update} - \ref{eq:nu_update}) called by ADMM determine the per iteration complexity.
    The update rules of the ADMM algorithm used to optimize the augmented split Lagrangian are
    \begin{align}
    \beta^{k+1} = \arg \min_{\beta \in P \times D} \mathcal L_{\rho,\gamma,\nu} (X,\beta,z^k) \\
    z^{k+1} = \arg \min_{z \in P \times D} \mathcal L_{\rho,\gamma,\nu} (X,\beta^{k+1},z) \\
    \nu^{k+1} = \nu^k + \rho (I_D - w(X) \beta^{k+1})
    \end{align}
    We will analyze the computational complexity of each of these steps by rewriting the update rules according to the specifics of the isometry pursuit problem.

The beta update Equation \ref{eq:beta_update} projects the solution onto the feasible set.
\begin{align}
    \label{eq:beta_update}
    \beta^{k+1} &= \arg \min_{\beta \in P \times D}  \text{tr}({\nu^k}^T (I_D - w(X) \beta)) + \rho \|I_D - w(X) \beta\|_2^2 + \gamma \|\beta - z^k\|_2^2 \\
    &= (\rho \, w(X)^T w(X) + \gamma I_D)^{-1} (\rho \, w(X)^T I_D + \gamma z^k + w(X)^T \nu^k)
    \end{align}
% This is accomplished through computation of the pseudo-inverse of the linear regression solution $(w(X)^T w(X))^{-1} w(X)^T I_D$.
The computational complexity of this computation is dominated by the inverse - that is, $O(D^2P)$.

Subsequently, the z update Equation \ref{eq:z_update} is soft-thresholding operation onto a contracted level set $\|\|_{1,2}$ ball.
\begin{align}
    \label{eq:z_update}
    z^{k+1} &= \arg \min_{z \in P \times D} \|z\|_{1,2} + \gamma \|z - \beta^{k+1}\|_2^2 \\
    &= \begin{cases} (1 - \frac{1}{\gamma\|\beta_{j.}^{k+1}\|_2}) (\beta_{j.}^{k+1}) \; \text{if} \; \|\beta_{j.}^{k+1}\|_2 > \frac{1}{\gamma}\\
        0 \; \text{otherwise}
    \end{cases}
\end{align}
The computational complexity of this operation is $O(PD)$.

Finally, the $\nu$ update Equation \ref{eq:nu_update} increases the squeezing of the $z$ and $\beta$ variables towards to each other.
\begin{align}
\nu^{k+1} = \nu^k + \rho (I_D - w(X) \beta^{k+1})
\label{eq:nu_update}
\end{align}
In this step, the dual variable $\nu$ that controls the penalty accrued by the splitting is updated in the direction of the constraint violation.
The computational complexity of this operation is dominated by that of the matrix multiplication $w(X) \beta^{k+1}$.
That is, it is $O(D^2P)$.
\end{proof}

\subsection{Partial derivatives of candidate losses}
\label{sec:loss_derivatives}

We summarize the column-wise derivatives of the losses in Table \ref{tab:loss_functions}.
Throughout this section let $X = [x_1,\dotsc,x_D] \in \mathbb R^{D \times D}$ be full rank.
For a scalar loss $L(X)$, write
\begin{align}
\nabla_X L = \left[
\frac{\partial L}{\partial x_1},
\dotsc,
\frac{\partial L}{\partial x_D}
\right].
\end{align}
All formulas are stated on smooth strata: rank deficient matrices, repeated singular values, and zero rows of pseudoinverse factors should be handled using subgradients.

We will use two common identities.
First, if $X = U \operatorname{diag}(\sigma) V^\top$ is a singular value decomposition and
$L(X) = \sum_{d=1}^D \phi(\sigma_d(X))$, then
\begin{align}
\nabla_X L = U \operatorname{diag}(\phi'(\sigma_1),\dotsc,\phi'(\sigma_D)) V^\top .
\label{eq:spectral_gradient_identity}
\end{align}
Second, for normalized losses let $N = n_q^{\times D}(X)$.
If $M = \nabla_N L$, then the column-wise chain rule is
\begin{align}
\frac{\partial L}{\partial x_p} = \mathrm{D}n_q(x_p)^\top M_{.p}.
\label{eq:column_chain_rule}
\end{align}
For $r=\|x\|_2$, write $n_q(x)=s(r)x$ with $s(r)=q(r)/r$.
Then
\begin{align}
\mathrm{D}n_q(x) &= s(r) I_D + \frac{s'(r)}{r}xx^\top, \\
s'(r) &= \frac{q'(r)}{r} - \frac{q(r)}{r^2}, \\
q'(r) &= -\frac{2e(e^r-r^{-2}e^{1/r})}{(e^r+e^{1/r})^2}.
\label{eq:normalization_jacobian}
\end{align}
Since $\mathrm{D}n_q(x)$ is symmetric, the transpose in this chain rule may be omitted.

\paragraph{Distortion.}
Let $\sigma_1=\sigma_{\max}(X)$ and $\sigma_D=\sigma_{\min}(X)$ with corresponding singular vectors $u_1,v_1$ and $u_D,v_D$.
For
\begin{align}
L_{\mathrm{dist}}(X)=\frac{\sigma_1}{\sigma_D}-1,
\end{align}
The spectral gradient identity gives
\begin{align}
\nabla_X L_{\mathrm{dist}}
= \frac{u_1v_1^\top}{\sigma_D}
- \frac{\sigma_1}{\sigma_D^2}u_Dv_D^\top .
\end{align}
The derivative is insensitive to uniform rescaling, which is the sense in which this loss does not prefer unit length.

\paragraph{Isometry loss.}
For
\begin{align}
L_{\mathrm{iso}}(X)=\sum_{d=1}^D g(\sigma_d(X))-D,
\end{align}
we have
\begin{align}
g'(t) = \frac{e^t - t^{-2}e^{1/t}}{2e}.
\end{align}
Thus
\begin{align}
\nabla_X L_{\mathrm{iso}}
= U \operatorname{diag}(g'(\sigma_1),\dotsc,g'(\sigma_D))V^\top .
\end{align}

\paragraph{S-divergence to identity.}
Let $G=X^\top X$ and
\begin{align}
L_{\mathrm{sdiv}}(X)
= \log\det\left(\frac{G+I_D}{2}\right)
-\frac{1}{2}\log\det(G).
\end{align}
The constant $\log\det(1/2)$ does not contribute to the derivative.
Using $d\log\det A = \operatorname{tr}(A^{-1}dA)$ and $dG=dX^\top X+X^\top dX$,
\begin{align}
\nabla_X L_{\mathrm{sdiv}}
= 2X(G+I_D)^{-1} - XG^{-1}.
\end{align}

\paragraph{Normalized log-det volume.}
Let $N=n_q^{\times D}(X)$ and
\begin{align}
L_{\mathrm{nld}}(X)= -\sum_{d=1}^D \log\sigma_d(N).
\end{align}
For square full-rank $N$, this is $-\log|\det N|$, so
\begin{align}
\nabla_N L_{\mathrm{nld}} = -N^{-\top}.
\end{align}
Therefore, by the column-wise chain rule,
\begin{align}
\frac{\partial L_{\mathrm{nld}}}{\partial x_p}
= -\mathrm{D}n_q(x_p)^\top (N^{-\top})_{.p}.
\end{align}

\paragraph{Frobenius pseudoinverse.}
Let $N=n_q^{\times D}(X)$ and $B=N^{-1}$.
For
\begin{align}
L_{\mathrm{frob}}(X)=\|N^{-1}\|_F-\sqrt{D},
\end{align}
the differential identity $dB=-B(dN)B$ gives
\begin{align}
\nabla_N L_{\mathrm{frob}}
= -\frac{N^{-\top}N^{-1}N^{-\top}}{\|N^{-1}\|_F}.
\end{align}
Thus
\begin{align}
\frac{\partial L_{\mathrm{frob}}}{\partial x_p}
= \mathrm{D}n_q(x_p)^\top
\left(
-\frac{N^{-\top}N^{-1}N^{-\top}}{\|N^{-1}\|_F}
\right)_{.p}.
\end{align}

\paragraph{Isometry pursuit proxy.}
Let $N=n_q^{\times D}(X)$ and $B=N^{-1}$.
The differentiable proxy for the \isometrypursuit~ table row is
\begin{align}
L_{\mathrm{ip}}(X)=\|N^{-1}\|_{1,2}-D.
\end{align}
Define $R \in \mathbb R^{D\times D}$ by
\begin{align}
R_{p.}=\frac{B_{p.}}{\|B_{p.}\|_2}.
\end{align}
Then $R \in \partial \|B\|_{1,2}$ on the smooth stratum where every row of $B$ is nonzero, and
\begin{align}
\nabla_N L_{\mathrm{ip}} = -B^\top R B^\top.
\end{align}
The column-wise gradient is therefore
\begin{align}
\frac{\partial L_{\mathrm{ip}}}{\partial x_p}
= \mathrm{D}n_q(x_p)^\top (-B^\top R B^\top)_{.p}.
\end{align}
